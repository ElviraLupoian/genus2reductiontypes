% III-II_l, l>1.
% The two principal components meet tangentially on the left. The chain on
% the right meets each of them transversely. It has l-1 components, giving
% 2+(l-1)=l+1 components in total.

\begin{tikzpicture}[
  x=1cm,
  y=1cm,
  line cap=round,
  line join=round,
  ordinary/.style={line width=0.4pt},
  principal/.style={line width=0.55pt}
]

  % Principal components.  On the left these are the actual quadratic
  % graphs y=+-0.345x^2, so their contact at the origin is unambiguous.
  \draw[principal,domain=-1.05:0,samples=80,smooth,variable=\x]
    plot (\x,{0.345*(\x)^2});
  \draw[principal,domain=-1.05:0,samples=80,smooth,variable=\x]
    plot (\x,{-0.345*(\x)^2});

  % A short C^1-style transition lets each parabola become a straight line
  % gradually instead of producing a corner at the origin.
  \draw[principal]
    (0,0)
      .. controls (0.12,0.00) and (0.30,0.17) ..
    (0.42,0.26)
      -- (1.05,0.74);
  \draw[principal]
    (0,0)
      .. controls (0.12,0.00) and (0.30,-0.17) ..
    (0.42,-0.26)
      -- (1.05,-0.74);

  \node[nu label] at (-0.56,0.34) {$B$};
  \node[nu label] at (-0.56,-0.34) {$B$};

  % The chain joining the two principal components. Its inner components
  % are suppressed in the usual MRNC style.
  \draw[ordinary] (0.68,0.62) -- (1.48,0.42);
  \fill (1.48,0.18) circle[radius=0.65pt];
  \fill (1.48,0.00) circle[radius=0.65pt];
  \fill (1.48,-0.18) circle[radius=0.65pt];
  \draw[ordinary] (1.48,-0.42) -- (0.68,-0.62);

  \node[nu annotation,anchor=west] at (1.63,0) {$l - 1$};

\end{tikzpicture}

\end{document}
